\section{Tổng quan sản phẩm}
BandUp là nền tảng AI Writing Coach hỗ trợ người học luyện IELTS Writing Task 2. Sản phẩm không chỉ trả về một band score ước lượng mà tạo một vòng lặp học tập dài hạn: viết bài, nhận feedback, hiểu lỗi, lưu lỗi, theo dõi tiến bộ và luyện tập lại.

\section{Phạm vi MVP}
\subsection{Trong phạm vi}
\begin{itemize}
  \item Google OAuth/\allowbreak{}OIDC kết hợp ASP.NET Core Identity và application cookie-session.
  \item Landing Page, bảng thông báo, FAQ, Trợ giúp \& Pháp lý.
  \item Onboarding band hiện tại, band mục tiêu, ngày thi dự kiến và mục tiêu học.
  \item Danh sách đề có đánh số, tìm kiếm/\allowbreak{}lọc, chọn đề, random đề và Redo Challenge.
  \item Writing Room với timer 40 phút, word count, autosave, pause/\allowbreak{}resume.
  \item Queue chấm bài, BandUp Credit reservation/\allowbreak{}capture/\allowbreak{}release và một model GPT-5.6 Terra.
  \item Kết quả chấm bốn tiêu chí, corrections, vocabulary suggestions, errors và next practice plan.
  \item Model Essays đã review, History, Mistake Bank, Progress và Redo comparison.
  \item Mua gói credit qua payOS/\allowbreak{}VietQR, webhook, đối soát và lịch sử thanh toán.
  \item In-app support ticket, tranh chấp kết quả, hoàn credit có kiểm soát.
  \item Admin quản lý nội dung, user, activity, credit, payment, revenue, AI usage, support, security và settings.
  \item Rate limit, budget cap, kill switch, logging, audit, backup và quy trình xóa tài khoản.
\end{itemize}
\subsection{Ngoài phạm vi MVP}
\begin{itemize}
  \item IELTS Speaking, Listening và Reading.
  \item AI Voice và mobile app native.
  \item Diễn đàn cộng đồng đầy đủ và gamification phức tạp.
  \item Teacher Classroom hoàn chỉnh.
  \item Flashcards và spaced repetition.
  \item Subscription định kỳ tự động.
  \item Pipeline chấm nhiều model hoặc model fallback tự động.
  \item Cam kết band score hoặc thay thế giám khảo IELTS.
\end{itemize}

\section{Giả định và ràng buộc}
\begin{itemize}
  \item Phiên bản đầu chỉ hỗ trợ IELTS Writing Task 2.
  \item IELTS Task 2 chính thức yêu cầu ít nhất 250 từ và khoảng 40 phút; BandUp cho phép 200-249 từ sau cảnh báo, 250-350 từ bình thường và chặn trên 350 từ.
  \item Band score từ 0 đến 9, bước 0.5; chỉ là ước lượng.
  \item User đăng nhập Google nhưng BandUp vẫn phát hành application cookie-session riêng.
  \item Free user nhận 5 credit mỗi chu kỳ miễn phí; mặc định chu kỳ một tháng và phải cấu hình được.
  \item Một grading thành công mặc định tốn 1 credit.
  \item OpenAI là provider ban đầu và GPT-5.6 Terra là model chấm chính duy nhất trong MVP.
  \item Development dùng local database/\allowbreak{}storage; production dự kiến container trên VPS.
  \item AI, payment, storage và email phải nằm sau interface; database dùng EF Core provider abstraction.
  \item Model Essays và payOS payment thuộc MVP.
  \item Xóa tài khoản xóa/\allowbreak{}anonymize dữ liệu học tập; payment/\allowbreak{}accounting và audit bắt buộc giữ tuân theo retention.
\end{itemize}

\subsection{Authentication, cookie-session, authorization và tài khoản}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-001}\hfill\priority{P0}\par
Hệ thống phải cho phép user đăng nhập bằng tài khoản Google thông qua OAuth 2.0/\allowbreak{}OIDC.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-002}\hfill\priority{P0}\par
BandUp không được yêu cầu hoặc lưu mật khẩu Google của user.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-003}\hfill\priority{P0}\par
Sau lần đăng nhập Google đầu tiên, hệ thống phải tạo một local User và liên kết với định danh Google ổn định của tài khoản.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-004}\hfill\priority{P0}\par
Hệ thống phải kiểm tra email và trạng thái email verified do Google cung cấp trước khi tạo/\allowbreak{}liên kết tài khoản theo chính sách.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-005}\hfill\priority{P0}\par
Đăng nhập lại bằng cùng một Google account phải liên kết đúng local User, không tạo tài khoản trùng.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-006}\hfill\priority{P0}\par
Hệ thống phải cho phép người dùng đăng xuất và thu hồi session BandUp.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-007}\hfill\priority{P0}\par
Hệ thống phải duy trì trạng thái đăng nhập bằng secure authentication cookie trong thời hạn session được cấu hình.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-008}\hfill\priority{P0}\par
Hệ thống phải từ chối tài khoản bị khóa, suspended hoặc deletion pending.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-009}\hfill\priority{P0}\par
Hệ thống phải phân quyền tối thiểu theo USER và ADMIN.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-010}\hfill\priority{P0}\par
Authentication chỉ xác định user là ai; authorization phải dùng role, claim, policy và ownership để quyết định user được làm gì.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-011}\hfill\priority{P0}\par
USER được sử dụng chức năng học tập, billing và support thuộc tài khoản của mình, nhưng không được truy cập Admin Portal hoặc dữ liệu của user khác.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-012}\hfill\priority{P0}\par
ADMIN được truy cập chức năng vận hành theo permission được cấp, nhưng không được bỏ qua ownership/\allowbreak{}privacy rule nếu permission đó không cần thiết.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-013}\hfill\priority{P0}\par
Các endpoint admin phải yêu cầu role/\allowbreak{}policy tương ứng; chỉ ẩn menu frontend không được xem là cơ chế bảo mật.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-014}\hfill\priority{P0}\par
Người dùng phải có thể xem và cập nhật hồ sơ cơ bản gồm tên hiển thị, avatar, band hiện tại, band mục tiêu, ngày thi dự kiến và cài đặt học tập.
\end{requirementbox}
\begin{requirementbox}[BandBlue]
\textbf{FR-AUTH-015}\hfill\priority{P0}\par
Hệ thống phải hỗ trợ yêu cầu xóa tài khoản và hiển thị rõ dữ liệu nào bị xóa, dữ liệu nào phải giữ tối thiểu vì nghĩa vụ pháp lý hoặc chống gian lận.

\textbf{When:} Backend nhận webhook có signature hợp lệ và trạng thái PAID.\par
\textbf{Then:} Payment chuyển PAID, credit được cộng đúng một lần và lịch sử hiển thị receipt.
\end{acceptancebox}

\section{Tiêu chí hoàn thành MVP}
\begin{itemize}
  \item Google Login + secure cookie-session + role/\allowbreak{}policy/\allowbreak{}ownership hoạt động.
  \item Landing, onboarding không ép làm bài, Dashboard và thông báo/\allowbreak{}pháp lý hoàn chỉnh.
  \item Prompt catalog đánh số, select/\allowbreak{}random/\allowbreak{}redo hoạt động.
  \item Writing Room, autosave, timer, word count và pause/\allowbreak{}resume đạt acceptance criteria.
  \item Submission + reservation/\allowbreak{}capture/\allowbreak{}release credit idempotent.
  \item Một model GPT-5.6 Terra chấm qua queue, structured output, tối đa 3 attempt và circuit breaker.
  \item Result, Model Essays, History, Mistake Bank, Progress và Redo hoạt động.
  \item payOS order, VietQR checkout, webhook verification, idempotency và reconciliation hoạt động.
  \item Support/\allowbreak{}refund ticket và admin handling có audit.
  \item Admin quản lý content, user, activity, credit, payment, revenue, AI usage, security và settings theo permission.
  \item Xóa tài khoản, retention, backup lifecycle và DeletionLedger được test.
  \item Functional, integration, payment, security, load-smoke, backup/\allowbreak{}restore và recovery test pass.
  \item Có thể deploy staging/\allowbreak{}production bằng container trên VPS và có runbook tối thiểu.
\end{itemize}
